chainlink/
├── Makefile                # Grouping the codes together
├── README.md
├── data/
│   └── classicmode.txt          # pre-made 7-word chains for "game master" lists
│   └── blitzmode.txt          # pre-made 30-word chains for "game master" lists
├── common/                 # shared by server and client
│   ├── protocol.h          # message types/format (e.g. HINT, TURN, GUESS, WIN)
│   ├── net.h               # C socket API declarations (extern "C" guarded)
│   └── net.c               # C: socket/bind/listen/accept/connect, send_line, recv_line
├── server/
│   ├── server.c               
│   └── client.c           
└── client/
    ├── main.cpp                # starts server, accepts 2 clients, runs a match
    ├── session.h / .cpp        # game master: coin flip, mode + list-source choice, broadcasts
    ├── game.h / .cpp           # core logic: Player, chain, turn switching, hints, win check
    ├── classic_mode.h / .cpp   # Classic rules (turn ends on wrong guess)
    ├── blitz_mode.h / .cpp     # Chain Blitz rules (30s timer per player, most words wins)
    └── word_bank.h / .cpp      # loads chains.txt, picks a random chain, validates player-made lists


server.c        networking only: socket, bind, listen, accept two players
   │            then hands both sockets to the session
   ▼
session.c       the "game master": asks setup questions, flips the coin,
   │            relays messages between the two sockets, announces the winner
   │
   ├── word_bank.c     loads data/*.json, picks a random chain, validates player lists
   ├── game.c          the chain, hints, guess checking, scores
   ├── classic_mode.c  Classic rules: wrong guess ends the turn
   └── blitz_mode.c    Blitz rules: 30s clock per player
